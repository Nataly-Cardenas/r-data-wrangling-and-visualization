2 Historia Olímpica: Rio 2016

url <- "https://raw.githubusercontent.com/cdeoroaguado/Datos/refs/heads/main/dataviz/athlete_events.csv"
datos_2 <- read.csv(url)

df_2016 <- datos_2 %>%
  filter(Year == 2016, !is.na(Medal))

2.1 Top 5 Deportes con Más Medallas en 2016

top5_deportes <- df_2016 %>%
  group_by(Sport) %>%
  summarise(numero_medallas = n(), .groups = "drop") %>%
  arrange(desc(numero_medallas)) %>%
  slice_head(n = 5)

top5_deportes %>% 
  gt() %>% 
  tab_header(
    title = md("**Deportes Principales en Río 2016**"),
    subtitle = "Top 5 según cantidad total de medallas concedidas"
  ) %>% 
  cols_label(
    Sport = "Deporte",
    numero_medallas = "N° de Medallas"
  ) %>% 
  tab_options(
    table.width = pct(60),
    column_labels.background.color = "#3D1534", # Midnight Violet
    column_labels.font.weight = "bold"
  )

2.2 Distribución de Edad de los Medallistas

deportes_top <- top5_deportes$Sport

tabla_dist <- df_2016 %>%
  filter(Sport %in% deportes_top) %>%
  group_by(Sport) %>%
  summarise(
    media = mean(Age, na.rm = TRUE),
    mediana = median(Age, na.rm = TRUE),
    desviacion = sd(Age, na.rm = TRUE),
    minimo = min(Age, na.rm = TRUE),
    maximo = max(Age, na.rm = TRUE),
    Q1 = quantile(Age, 0.25, na.rm = TRUE),
    Q3 = quantile(Age, 0.75, na.rm = TRUE),
    RIC = IQR(Age, na.rm = TRUE),
    .groups = "drop"
  )

tabla_dist %>% 
  gt() %>% 
  tab_header(title = md("**Resumen Estadístico de Edad por Deporte**")) %>% 
  cols_label(
    Sport = "Deporte", media = "Media", mediana = "Mediana",
    desviacion = "DE", minimo = "Mín", maximo = "Máx",
    Q1 = "Q1", Q3 = "Q3", RIC = "RIC"
  ) %>% 
  fmt_number(columns = where(is.numeric), decimals = 1) %>% 
  tab_options(
    table.width = pct(100), 
    column_labels.background.color = "#3E4B8E"
  )

ggplot(df_2016 %>% filter(Sport %in% deportes_top), aes(x = reorder(Sport, Age, FUN = median), y = Age, fill = Sport)) +
  geom_boxplot(alpha = 0.85, color = "#3D1534", outlier.color = "#3E4B8E", outlier.size = 2) +
  scale_fill_manual(values = unname(paleta_custom)) +
  labs(
    title = "Distribución de edad en medallistas de Río 2016",
    subtitle = "Comparación entre los 5 deportes principales",
    x = "Deporte",
    y = "Edad (años)"
  ) +
  theme_minimal(base_size = 12) +
  theme(
    plot.title = element_text(face = "bold", size = 14, color = "#3D1534"),
    legend.position = "none",
    panel.grid.minor = element_blank()
  )

2.3 Equipos Líderes por Deporte

tabla_equipos <- df_2016 %>%

  ungroup()

2.4 Análisis del Peso por Sexo

tabla_peso <- df_2016 %>%
  filter(Sport %in% deportes_top) %>%
  group_by(Sport, Sex) %>%
  summarise(
    media = mean(Weight, na.rm = TRUE),
    mediana = median(Weight, na.rm = TRUE),
    RIC = IQR(Weight, na.rm = TRUE),
    .groups = "drop"
  )

tabla_peso %>% 
  gt() %>% 
  tab_header(title = md("**Tendencia de Peso por Deporte y Sexo**")) %>% 
  cols_label(
    Sport = "Deporte", Sex = "Sexo", media = "Media (kg)",
    mediana = "Mediana (kg)", RIC = "RIC"
  ) %>% 
  fmt_number(columns = c(media, mediana, RIC), decimals = 1) %>% 
  tab_options(
    table.width = pct(80), 
    column_labels.background.color = "#3E4B8E"
  )
